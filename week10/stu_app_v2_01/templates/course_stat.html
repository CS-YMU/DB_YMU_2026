存储过程源码： 
CREATE PROCEDURE sp_CourseStat(
    IN  p_cid INT,              -- 输入：课程编号
    OUT out_avg_grade DECIMAL(5,2), -- 输出：平均分
    OUT out_max_grade DECIMAL(5,2), -- 输出：最高分
    OUT out_min_grade DECIMAL(5,2), -- 输出：最低分
    OUT out_pass_num INT,          -- 输出：及格人数(成绩≥60)
    OUT out_total_num INT          -- 输出：选修总人数
)
BEGIN
    SELECT IFNULL(AVG(Grade), 0), IFNULL(MAX(Grade), 0),
           IFNULL(MIN(Grade), 0),
           IFNULL(SUM(CASE WHEN Grade >= 60 THEN 1 ELSE 0 END), 0),
           COUNT(*)
    INTO out_avg_grade, out_max_grade, out_min_grade,
         out_pass_num, out_total_num
    FROM student_course WHERE CourseID = p_cid;
END
调用方式：CALL sp_CourseStat(1, @avg, @max, @min, @pass, @total);
